set(QJS "${CMAKE_SOURCE_DIR}/.deps/quickjs-2026-06-04")
add_library(c1quickjs STATIC "${QJS}/quickjs.c" "${QJS}/dtoa.c" "${QJS}/libregexp.c" "${QJS}/libunicode.c" "${QJS}/cutils.c")
# Only export the API header: QuickJS's VERSION file shadows C++ <version>
# when the source lives on a case-insensitive host filesystem.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include")
configure_file("${QJS}/quickjs.h" "${CMAKE_CURRENT_BINARY_DIR}/include/quickjs.h" COPYONLY)
target_include_directories(c1quickjs PRIVATE "${QJS}" PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/include")
target_compile_definitions(c1quickjs PRIVATE _GNU_SOURCE CONFIG_VERSION="2026-06-04")
target_compile_options(c1quickjs PRIVATE -fwrapv)
target_link_libraries(c1quickjs m atomic)
add_library(c1sketch STATIC src/runtime.cpp)
target_include_directories(c1sketch PUBLIC src)
target_link_libraries(c1sketch PUBLIC c1quickjs)
add_executable(c1max-sketch-test tests/runtime_test.cpp)
target_link_libraries(c1max-sketch-test c1sketch)
target_compile_options(c1max-sketch-test PRIVATE -UNDEBUG)
set_target_properties(c1max-sketch-test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_executable(c1max-processing src/main.cpp ../shared/display.cpp ../shared/keyboard.cpp)
target_link_libraries(c1max-processing c1sketch c1net lvgl m)
set_target_properties(c1max-processing PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
