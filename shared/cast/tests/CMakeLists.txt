cmake_minimum_required(VERSION 3.16)
project(cast_protocol_checks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
add_subdirectory(${ROOT}/.deps/mbedtls-2.28.10 mbedtls EXCLUDE_FROM_ALL)
add_library(protocol STATIC ../wire.cpp ../discovery.cpp ../chromecast.cpp ../dlna.cpp ${ROOT}/crosspoint/vendor/tinyxml2/tinyxml2.cpp)
target_include_directories(protocol PUBLIC ${ROOT}/.build/include PRIVATE ${ROOT}/crosspoint/vendor/tinyxml2)
target_link_libraries(protocol mbedtls mbedx509 mbedcrypto)
add_executable(protocol-test protocol_test.cpp)
target_link_libraries(protocol-test protocol)
target_compile_options(protocol-test PRIVATE -UNDEBUG)
add_executable(protocol-probe probe.cpp)
target_link_libraries(protocol-probe protocol)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(daemon-test ../daemon.cpp ../../cast_client.cpp ../../net.cpp fake_discovery.cpp)
    target_include_directories(daemon-test PRIVATE ${ROOT}/shared)
    target_link_libraries(daemon-test protocol pthread)
endif()
