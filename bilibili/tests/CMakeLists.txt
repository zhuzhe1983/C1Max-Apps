cmake_minimum_required(VERSION 3.16)
project(c1_bilibili_cast_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(APPS "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
add_subdirectory("${APPS}/.deps/mbedtls-2.28.10" "${CMAKE_BINARY_DIR}/mbedtls" EXCLUDE_FROM_ALL)
add_library(test_api STATIC "${APPS}/bilibili/src/api.cpp" "${APPS}/shared/net.cpp")
target_include_directories(test_api PUBLIC "${APPS}/bilibili/src" "${APPS}/shared" "${APPS}/.build/include")
target_link_libraries(test_api PUBLIC mbedcrypto pthread)
add_executable(relay-probe relay_probe.cpp "${APPS}/bilibili/src/cast_relay.cpp")
target_link_libraries(relay-probe PRIVATE test_api)
add_executable(player-cast-test player_cast_test.cpp "${APPS}/bilibili/src/player.cpp")
target_include_directories(player-cast-test PRIVATE "${APPS}" "${APPS}/.deps/lvgl" "${APPS}/streamplayer/src")
target_compile_definitions(player-cast-test PRIVATE LV_CONF_INCLUDE_SIMPLE)
target_link_libraries(player-cast-test PRIVATE test_api)
foreach(target test_api relay-probe player-cast-test)
    target_compile_options(${target} PRIVATE -UNDEBUG)
endforeach()
