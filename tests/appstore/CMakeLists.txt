cmake_minimum_required(VERSION 3.16)
project(c1store_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT ../.. ABSOLUTE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
add_subdirectory(${ROOT}/.deps/mbedtls-2.28.10 ${CMAKE_BINARY_DIR}/mbedtls EXCLUDE_FROM_ALL)
add_library(c1store STATIC ${ROOT}/appstore/src/store.cpp ${ROOT}/shared/net.cpp ${ROOT}/shared/audio_client.c)
target_include_directories(c1store PUBLIC ${ROOT}/appstore/src ${ROOT}/shared ${ROOT}/.build/include)
target_link_libraries(c1store PUBLIC mbedcrypto pthread)
add_executable(store-test store_test.cpp)
target_compile_options(store-test PRIVATE -UNDEBUG)
target_link_libraries(store-test c1store)
set(LV_BUILD_CONF_PATH "${ROOT}/shared/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
add_subdirectory(${ROOT}/.deps/lvgl ${CMAKE_BINARY_DIR}/lvgl EXCLUDE_FROM_ALL)
add_executable(store-preview ${ROOT}/appstore/src/main.cpp ${ROOT}/tests/tox/headless.cpp)
target_include_directories(store-preview PRIVATE ${ROOT}/.deps/lvgl/src/libs/tiny_ttf)
target_link_libraries(store-preview c1store lvgl m)
