set(DBP "${CMAKE_CURRENT_BINARY_DIR}/vendor")
execute_process(COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare.py" "${CMAKE_SOURCE_DIR}/.deps/dosbox-pure" "${DBP}" RESULT_VARIABLE prepared)
if(NOT prepared EQUAL 0)
    message(FATAL_ERROR "DOSBox source preparation failed")
endif()
file(GLOB DBP_SOURCES "${DBP}/*.cpp" "${DBP}/src/*.cpp" "${DBP}/src/*/*.cpp" "${DBP}/src/*/*/*.cpp")
add_library(c1doscore STATIC ${DBP_SOURCES})
target_include_directories(c1doscore PRIVATE "${DBP}/include")
target_compile_definitions(c1doscore PRIVATE __LIBRETRO__ _FILE_OFFSET_BITS=64 NDEBUG DISABLE_DYNAREC)
target_compile_options(c1doscore PRIVATE -O2 -fno-strict-aliasing -Wno-address-of-packed-member -Wno-format -Wno-psabi)
target_link_libraries(c1doscore PUBLIC pthread m atomic)
add_executable(c1max-dos-core src/core.cpp ../launcher/src/typeface.c)
target_include_directories(c1max-dos-core PRIVATE "${CMAKE_SOURCE_DIR}/.deps/lvgl/src/libs/tiny_ttf")
target_include_directories(c1max-dos-core PRIVATE "${DBP}/libretro-common/include")
target_link_libraries(c1max-dos-core c1doscore tinyalsa)
set_target_properties(c1max-dos-core PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_executable(c1max-dosbox src/library.cpp ../shared/display.cpp ../shared/keyboard.cpp)
target_link_libraries(c1max-dosbox c1net lvgl m)
set_target_properties(c1max-dosbox PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
find_program(NASM nasm REQUIRED)
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/C1LAB.COM"
  COMMAND "${NASM}" -f bin "${CMAKE_CURRENT_SOURCE_DIR}/tests/lab.asm" -o "${CMAKE_BINARY_DIR}/C1LAB.COM"
  DEPENDS tests/lab.asm VERBATIM)
add_custom_target(c1max-dos-lab ALL DEPENDS "${CMAKE_BINARY_DIR}/C1LAB.COM")
add_executable(c1max-dos-video-test tests/video_test.cpp)
target_compile_options(c1max-dos-video-test PRIVATE -UNDEBUG)
set_target_properties(c1max-dos-video-test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
